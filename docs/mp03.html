# Task 3: Download, unzip, and load NYC 2020 NTA boundaries
load_nyc_nta <- function() {
  dir.create("data/mp03", recursive = TRUE, showWarnings = FALSE)
  
  nta_zip <- "data/mp03/nynta2020_shp.zip"
  nta_dir <- "data/mp03/nynta2020_shp"
  nta_url <- "https://s-media.nyc.gov/agencies/dcp/assets/files/zip/data-tools/bytes/neighborhood-tabulation-areas/nynta2020_26d.zip"
  
  if (!file.exists(nta_zip)) {
    download.file(nta_url, destfile = nta_zip, mode = "wb")
  }
  
  if (!dir.exists(nta_dir)) {
    unzip(nta_zip, exdir = nta_dir)
  }

  shp_file <- list.files(nta_dir, pattern = "\\.shp$", full.names = TRUE, recursive = TRUE)[1]
  
  nyc_nta <- read_sf(shp_file) |>
    select(BoroCode, BoroName, CountyFIPS, NTA2020, NTAName, NTAType, geometry)
  
  return(nyc_nta)
}

# Task 5: Download PLUTO
load_nyc_pluto <- function() {
  dir.create("data/mp03", recursive = TRUE, showWarnings = FALSE)
  pluto_file <- "data/mp03/nyc_pluto.csv"
  
  if (!file.exists(pluto_file)) {
    base_url <- "https://data.cityofnewyork.us/resource/64uk-42ks.csv"
    select_cols <- paste(
      c(
        "bbl", "borough", "bct2020", "landuse", "bldgclass",
        "unitsres", "unitstotal", "resarea", "officearea",
        "comarea", "bldgarea", "numfloors"
      ),
      collapse = ","
    )
    
    limit <- 25000
    offset <- 0
    pages <- list()
    page_idx <- 1
    
    repeat {
      resp <- request(base_url) |>
        req_url_query(
          `$select` = select_cols,
          `$order`  = "bbl",
          `$limit`  = format(limit, scientific = FALSE),
          `$offset` = format(offset, scientific = FALSE)
        ) |>
        req_retry(max_tries = 10, backoff = ~ 3) |>
        req_perform()
      
      chunk <- read_csv(
        resp_body_string(resp),
        col_types = cols(.default = col_character()),
        show_col_types = FALSE
      )
      
      if (nrow(chunk) == 0) break
      
      pages[[page_idx]] <- chunk
      if (nrow(chunk) < limit) break
      
      offset <- offset + limit
      page_idx <- page_idx + 1
      Sys.sleep(0.5)
    }
    
    pluto_df <- bind_rows(pages) |>
      mutate(
        across(
          c(unitsres, unitstotal, resarea, officearea, comarea, bldgarea, numfloors),
          as.numeric
        )
      )
    
    write_csv(pluto_df, pluto_file)
  }
  
  read_csv(
    pluto_file,
    col_types = cols(
      bbl = col_character(),
      borough = col_character(),
      bct2020 = col_character(),
      landuse = col_character(),
      bldgclass = col_character(),
      .default = col_double()
    ),
    show_col_types = FALSE
  )
}

Historical Roots and the Journey to New York

Chinese immigration to the United States accelerated in the mid-nineteenth century during the California Gold Rush and the construction of the First Transcontinental Railroad. Despite their role in building western infrastructure, early immigrants faced severe legal discrimination, which led to the Chinese Exclusion Act of 1882—the first federal law barring entry based on nationality. Facing hostility in the West, many migrants moved east and established an enclave along Mott, Pell, and Doyers Streets in Lower Manhattan. After the Immigration and Nationality Act of 1965 lifted national-origins quotas, New York City’s Chinese population grew rapidly and expanded far beyond Manhattan.

Geographic Footprint in New York City

Today, New York City is home to 625,787 Chinese-American residents, accounting for 11.66% of all 5,368,065 Chinese-Americans nationwide. Rather than remaining centered in Manhattan’s historic Chinatown, the community now is concentrated in major residential and commercial neighborhoods in Brooklyn and Queens, particularly Sunset Park, Bensonhurst, Flushing, and Elmhurst. At the tract level, Census Tract 171.02; Queens County has the highest concentration in the city, where 90.70% of residents identify as Chinese-American.

load_nyc_nta <- function() {
  dir.create("data/mp03", recursive = TRUE, showWarnings = FALSE)
  
  nta_zip <- "data/mp03/nynta2020_shp.zip"
  nta_dir <- "data/mp03/nynta2020_shp"
  nta_url <- "https://s-media.nyc.gov/agencies/dcp/assets/files/zip/data-tools/bytes/neighborhood-tabulation-areas/nynta2020_26d.zip"
  
  if (!file.exists(nta_zip)) {
    download.file(nta_url, destfile = nta_zip, mode = "wb")
  }
  
  if (!dir.exists(nta_dir)) {
    unzip(nta_zip, exdir = nta_dir)
  }

  shp_file <- list.files(nta_dir, pattern = "\\.shp$", full.names = TRUE, recursive = TRUE)[1]
  
  nyc_nta <- read_sf(shp_file) |>
    select(BoroCode, BoroName, CountyFIPS, NTA2020, NTAName, NTAType, geometry)
  
  return(nyc_nta)
}

load_nyc_pluto <- function() {
  dir.create("data/mp03", recursive = TRUE, showWarnings = FALSE)
  pluto_file <- "data/mp03/nyc_pluto.csv"
  
  if (!file.exists(pluto_file)) {
    base_url <- "https://data.cityofnewyork.us/resource/64uk-42ks.csv"
    select_cols <- paste(
      c(
        "bbl", "borough", "bct2020", "landuse", "bldgclass",
        "unitsres", "unitstotal", "resarea", "officearea",
        "comarea", "bldgarea", "numfloors"
      ),
      collapse = ","
    )
    
    limit <- 25000
    offset <- 0
    pages <- list()
    page_idx <- 1
    
    repeat {
      resp <- request(base_url) |>
        req_url_query(
          `$select` = select_cols,
          `$order`  = "bbl",
          `$limit`  = format(limit, scientific = FALSE),
          `$offset` = format(offset, scientific = FALSE)
        ) |>
        req_retry(max_tries = 10, backoff = ~ 3) |>
        req_perform()
      
      chunk <- read_csv(
        resp_body_string(resp),
        col_types = cols(.default = col_character()),
        show_col_types = FALSE
      )
      
      if (nrow(chunk) == 0) break
      
      pages[[page_idx]] <- chunk
      if (nrow(chunk) < limit) break
      
      offset <- offset + limit
      page_idx <- page_idx + 1
      Sys.sleep(0.5)
    }
    
    pluto_df <- bind_rows(pages) |>
      mutate(
        across(
          c(unitsres, unitstotal, resarea, officearea, comarea, bldgarea, numfloors),
          as.numeric
        )
      )
    
    write_csv(pluto_df, pluto_file)
  }
  
  read_csv(
    pluto_file,
    col_types = cols(
      bbl = col_character(),
      borough = col_character(),
      bct2020 = col_character(),
      landuse = col_character(),
      bldgclass = col_character(),
      .default = col_double()
    ),
    show_col_types = FALSE
  )
}

#### Historical Roots and the Journey to New York

Chinese immigration to the United States accelerated in the mid-nineteenth century during the California Gold Rush and the construction of the First Transcontinental Railroad. Despite their role in building western infrastructure, early immigrants faced severe legal discrimination, which led to the **Chinese Exclusion Act of 1882**—the first federal law barring entry based on nationality. Facing hostility in the West, many migrants moved east and established an enclave along Mott, Pell, and Doyers Streets in Lower Manhattan. After the **Immigration and Nationality Act of 1965** lifted national-origins quotas, New York City’s Chinese population grew rapidly and expanded far beyond Manhattan.

#### Geographic Footprint in New York City

Today, New York City is home to **`r scales::comma(nyc_total_chinese)` Chinese-American residents**, accounting for **`r scales::percent(nyc_share_of_us, accuracy = 0.01)`** of all **`r scales::comma(us_total_chinese)`** Chinese-Americans nationwide. Rather than remaining centered in Manhattan’s historic Chinatown, the community now is concentrated in major residential and commercial neighborhoods in Brooklyn and Queens, particularly Sunset Park, Bensonhurst, Flushing, and Elmhurst.
